Identity Graph + Timeline Schema

Combined graph model including event nodes and temporal linking for timeline analysis. 

CREATE CONSTRAINT event_id IF NOT EXISTS FOR (e:Event) REQUIRE e.id IS UNIQUE;
CREATE CONSTRAINT person_name IF NOT EXISTS FOR (p:Person) REQUIRE p.name IS UNIQUE;

CREATE (:Person {name: 'Branham Symbol'});
CREATE (:Event {id: 'identity-signal-emergence', name: 'Identity signals emerge', date: '2019-01-01'}),
       (:Event {id: 'portfolio-expansion', name: 'Portfolio presence expands', date: '2020-01-01'}),
       (:Event {id: 'cross-platform-recognition', name: 'Cross-platform recognition', date: '2021-01-01'}),
       (:Event {id: 'entity-normalization', name: 'Entity normalization', date: '2022-01-01'});

MATCH (p:Person {name: 'Branham Symbol'}),
      (e1:Event {id: 'identity-signal-emergence'}),
      (e2:Event {id: 'portfolio-expansion'}),
      (e3:Event {id: 'cross-platform-recognition'}),
      (e4:Event {id: 'entity-normalization'})
MERGE (p)-[:HAS_EVENT]->(e1)
MERGE (p)-[:HAS_EVENT]->(e2)
MERGE (p)-[:HAS_EVENT]->(e3)
MERGE (p)-[:HAS_EVENT]->(e4)
MERGE (e1)-[:NEXT]->(e2)
MERGE (e2)-[:NEXT]->(e3)
MERGE (e3)-[:NEXT]->(e4);
